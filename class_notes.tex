\section{The Philosophy of Shoestring AI: Why Run Locally?}
\begin{itemize}
    \item \textbf{The Subscription Black Hole}: Commercial SaaS models lock basic capabilities behind \$20/month per-user paywalls. Running locally restores zero-marginal-cost computing.
    \item \textbf{Token Anxiety \& Financial Predictability}: Cloud APIs meter prompt tokens, completion tokens, and context caching. A runaway while-loop or bad agent recursive call can drain hundreds of dollars before dawn. Local inference costs \$0.00 per token.
    \item \textbf{Absolute Privacy \& Compliance}:
    \begin{itemize}
        \item Proprietary source code, customer records, and personal documents never touch third-party servers.
        \item Built-in immunity to model retraining on your private intellectual property.
    \end{itemize}
    \item \textbf{Zero Latency Jitter \& Offline Autonomy}: Functions in planes, trains, air-gapped server closets, or remote field sites without an internet connection.
\end{itemize}

\section{Narrow vs. Broad AI: Right-Sizing Your Model}
\begin{itemize}
    \item \textbf{Broad / Frontier AI (70B -- 1T+ parameters)}:
    \begin{itemize}
        \item General-purpose behemoths trained on trillions of tokens.
        \item Exceptional at high-abstract reasoning, multi-lingual poetry, complex thesis synthesis.
        \item \textbf{The Penalty}: Demands multi-GPU clusters (8x H100s, 640GB+ VRAM) or massive cloud rental bills.
    \end{itemize}
    \item \textbf{Narrow / Small Language Models (SLMs, 1B -- 8B parameters)}:
    \begin{itemize}
        \item Specialized powerhouses: \textbf{Phi-3.5} (Microsoft), \textbf{Qwen-2.5} (Alibaba), \textbf{SmolLM2} (Hugging Face), \textbf{Llama-3.2} (Meta).
        \item With domain fine-tuning and task-specific instruction prompts, a 3B--7B model frequently matches or beats a 70B model on parsing, JSON schema extraction, and shell command generation.
        \item \textbf{The Shoestring Rule}: Pick the smallest model that reliably satisfies your acceptance tests. Running a 70B model to lowercase a string is architectural malpractice.
    \end{itemize}
\end{itemize}

\section{AI Model Anatomy \& Weight Formats}
\begin{itemize}
    \item \textbf{What is a Model File?} A collection of neural network weight tensors ($\mathbf{W}$) and bias vectors ($\mathbf{b}$), accompanied by architectural metadata (layer count, attention head dimensions, context window length) and a tokenizer vocabulary.
    \item \textbf{The Primary File Formats}:
    \begin{itemize}
        \item \textbf{Safetensors (\texttt{.safetensors})}:
        \begin{itemize}
            \item Hugging Face open standard replacing insecure PyTorch \texttt{.bin}/\texttt{.pt} pickle files.
            \item Zero arbitrary code execution risk, zero deserialization overhead, fast memory-mapping (\texttt{mmap}). Default for training and fine-tuning.
        \end{itemize}
        \item \textbf{GGUF (GPT-Generated Unified Format)}:
        \begin{itemize}
            \item The undisputed universal format of consumer local inference, created by Georgi Gerganov (\texttt{llama.cpp}).
            \item \textbf{Self-Contained}: Contains all model architecture metadata, tokenizer configuration, and quantized weights in a single binary file.
            \item Supports split offloading between CPU system RAM and GPU VRAM seamlessly.
        \end{itemize}
        \item \textbf{ONNX \& EXL2}:
        \begin{itemize}
            \item \textbf{ONNX}: Standard for cross-platform edge inference (browsers, mobile, C\#/.NET).
            \item \textbf{EXL2}: Ultra-fast variable-bitrate quantization designed exclusively for high-throughput Nvidia GPUs via ExLlamaV2.
        \end{itemize}
    \end{itemize}
\end{itemize}

% SECTION 4: QUANTIZATION
% ----------------------------------------------------
\section{Model Quantization: Squeezing Giants into Laptops}
\begin{itemize}
    \item \textbf{Core Concept}: Neural networks are trained in 16-bit floating point (\texttt{FP16} = 2 bytes per weight). Quantization maps these continuous floats onto discrete low-bit integer bins (8-bit, 4-bit, or 2-bit).
    \item \textbf{Why it Works}: Neural network activations are intrinsically tolerant to stochastic noise. Reducing precision by 75\% (from 16-bit to 4-bit) typically increases model perplexity (error) by less than 1.5\%.
    \item \textbf{VRAM / RAM Calculation Formula}:
    \[
    \text{Required Memory (GB)} \approx \text{Parameter Count (in Billions)} \times \frac{\text{Quant Bit-Depth}}{8} \times 1.25 \quad (\text{includes KV cache overhead})
    \]
\end{itemize}

\small
\begin{tabular}{lllll}
\toprule

\midrule
\texttt{FP16} & 16.0 bits & 14.0 GB & Baseline (0.00) & Only for GPU billionaires \\

\section{The Local Runtimes Trinity: llama.cpp, Ollama, LM Studio}
\begin{itemize}
    \item \textbf{1. llama.cpp (The Engine Room)}:
    \begin{itemize}
        \item Written in pure C/C++ with SIMD optimizations (AVX2, AVX-512, Apple Metal, CUDA, ROCm, Vulkan).
        \item Zero dependencies, compiles in seconds with \texttt{make} or \texttt{cmake}.
        \item Supports fine-grained layer offloading: e.g., \texttt{-ngl 24} offloads 24 layers to GPU and keeps the rest in CPU RAM.
    \end{itemize}
    \item \textbf{2. Ollama (The Docker of AI)}:
    \begin{itemize}
        \item Wraps \texttt{llama.cpp} in a slick background daemon and CLI manager.
        \item Automatically detects GPU capabilities, pulls quantized GGUF models, and manages memory.
        \item Exposes a native REST API and an \textbf{OpenAI-compatible server} on port \texttt{11434}.
    \end{itemize}
    \item \textbf{3. LM Studio (The Visual Powerhouse)}:
    \begin{itemize}
        \item Desktop GUI (Electron/native) with built-in Hugging Face model search and downloader.
        \item Real-time generation token/sec stats, GPU VRAM allocation slider, and local developer server.
    \end{itemize}
\end{itemize}

% SECTION 7: DENSE VS MOE
% ----------------------------------------------------
\section{Architectures: Dense Models vs. Mixture of Experts (MoE)}
\begin{itemize}
    \item \textbf{Dense Models}: Every token passes through 100\% of the feed-forward network weights. (e.g., Llama-3-8B executes all 8 billion parameters on every word).
    \item \textbf{Sparse Mixture of Experts (MoE)}:
    \begin{itemize}
        \item The dense feed-forward layer is divided into $E$ distinct expert sub-networks (e.g., 8, 16, or 64 experts).
        \item A tiny, learned \textbf{Router Network} evaluates the token embedding and routes it to only the Top-$K$ experts (typically $K=2$).
        \item \textbf{Example}: Mixtral 8x7B has 47B total parameters, but only activates $\sim$13B parameters per token.
        \item \textbf{The Architectural Epiphany}: Since 85\%--95\% of weights sit idle for any given token, \textbf{why must they all reside in expensive VRAM?}
    \end{itemize}
\end{itemize}

\section{Spotlight Study: The Colibr\`i Project (Extreme Shoestring)}
\begin{colibribox}[The Colibr\`i Architecture: Pure C MoE Engine]
    \textbf{The Mission}: Shatter the assumption that running a 744-Billion parameter frontier model requires an enterprise datacenter rack.
    \begin{itemize}
        \item \textbf{Language}: Handcrafted in pure C for raw memory and disk I/O control.
        \item \textbf{Hardware Target}: A standard \$800 consumer laptop with only \textbf{24GB--25GB of RAM} and no dedicated GPU server rack.
        \item \textbf{The 3-Tier Memory Hierarchy}:
        \begin{enumerate}
            \item \textbf{Tier 1: System RAM ($\sim$10--15 GB)}: Pins the shared dense attention layers, token embeddings, normalization weights, and router permanently into fast memory.
            \item \textbf{Tier 2: High-Speed NVMe SSD ($\sim$370 GB)}: Houses the tens of thousands of routed expert parameters compressed in 4-bit quantization on disk.
            \item \textbf{Tier 3: Dynamic On-Demand Streaming + LRU Cache}: As the model decodes token-by-token, the router identifies the 2 active experts. Colibr\`i streams those exact weights from the NVMe SSD into an LRU cache in RAM in real time!
        \end{enumerate}
        \item \textbf{Performance vs. Freedom}:
        \begin{itemize}
            \item Generation speed ranges from \textbf{0.5 to 1.8 tokens/sec}.
            \item Is it instant? No. But it performs complex, frontier-level document analysis, code audit, and research locally, with zero cloud dependency and zero telemetry.
        \end{itemize}
    \end{itemize}
\end{colibribox}

\section{Local Autonomous Agents \& Tool Use (ReAct)}
\begin{itemize}
    \item \textbf{The ReAct (Reason + Act) Loop}:
    \begin{enumerate}
        \item \textbf{Prompt Injection}: User provides a high-level goal and available Python tools.
        \item \textbf{Reasoning}: Model outputs structured tool intent: \texttt{\{"tool": "query\_db", "args": \{"id": 42\}\}}.
        \item \textbf{Execution}: Python executes the function locally in the sandbox.
        \item \textbf{Feedback}: Tool observation is appended to message history.
        \item \textbf{Resolution}: Model evaluates whether the goal is completed or another action is required.
    \end{enumerate}
    \item \textbf{Deterministic JSON with Pydantic}: Small models can drift in formatting. Enforce strict JSON output using Ollama's schema validator:
\begin{lstlisting}[language=Python]
from pydantic import BaseModel

class ToolCall(BaseModel):
    command: str
    target_file: str

res = ollama.chat(
    model="qwen2.5:3b",
    messages=[...],
    format=ToolCall.model_json_schema()
)
\end{lstlisting}
    \item \textbf{Zero-Cost Loops}: Multi-step agent problem solving that would cost \$5 in cloud tokens costs nothing locally.
\end{itemize}

\section{Complete End-to-End Demo: Prompt to Result}
\begin{lstlisting}[language=Python]
import json, ollama
from pydantic import BaseModel, Field

# 1. Define Strict Pydantic Schema
class ServerIncident(BaseModel):
    service: str = Field(description="Name of crashed service")
    error_code: int = Field(description="HTTP or kernel error code")
    recommended_action: str = Field(description="Actionable fix")

# 2. Raw Unstructured Log Input
raw_log = """
[ERROR] 2026-10-16 21:05:12.341 [auth_service] worker-3 crashed!
Code: 504 Gateway Timeout while contacting Redis cluster at 10.0.4.12.
Recommend immediate restart of sentinel node.
"""

# 3. Local Model Structured Extraction
response = ollama.chat(
    model="qwen2.5:3b",
    messages=[
        {"role": "system", "content": "Extract server incident reports into strict JSON."},
        {"role": "user", "content": raw_log}
    ],
    format=ServerIncident.model_json_schema()
)

# 4. Validate and Execute Tool in Python
incident = ServerIncident.model_validate_json(response['message']['content'])
print(f"AUTOMATED ALERT: {incident.service} (Code {incident.error_code})")
print(f"ACTION REQUIRED: {incident.recommended_action}")
\end{lstlisting}

\begin{gotcha}[Three Deadly Sins of Local AI]
    \begin{enumerate}
        \item \textbf{The 128k Context Trap}: Expanding context window to 128k tokens allocates 8GB+ purely for the KV-cache. Keep default context to 4k or 8k unless needed.
        \item \textbf{Unquantized Weights (\texttt{FP16})}: Attempting to run raw FP16 weights on consumer hardware triggers memory thrashing, swap thrashing, and frozen laptops.
        \item \textbf{Benchmark Paralyzation}: Spending 40 hours benchmarking 15 different 7B models instead of building your actual agent workflow. Pick \texttt{Qwen2.5} or \texttt{Llama3.1} and start shipping!
    \end{enumerate}
\end{gotcha}
